-- Create Tables
CREATE TABLE states (
    sid INTEGER PRIMARY KEY,
    stname TEXT NOT NULL,
    stabbrev TEXT NOT NULL
);

CREATE TABLE categories (
    id INTEGER PRIMARY KEY,
    cname TEXT NOT NULL,
    cdescr TEXT
);

CREATE TABLE customers (
    id INTEGER PRIMARY KEY,
    fname TEXT NOT NULL,
    lname TEXT NOT NULL,
    state INTEGER NOT NULL REFERENCES states(sid)
);

CREATE TABLE products (
    id INTEGER PRIMARY KEY,
    pname TEXT NOT NULL,
    price NUMERIC(10, 2) NOT NULL CHECK (price >= 0),
    cid INTEGER NOT NULL REFERENCES categories(id)
);

CREATE TABLE sales (
    id INTEGER PRIMARY KEY,
    pid INTEGER NOT NULL REFERENCES products(id),
    cid INTEGER NOT NULL REFERENCES customers(id),
    quantity INTEGER NOT NULL CHECK (quantity >= 0),
    discount NUMERIC(5, 2) NOT NULL CHECK (discount >= 0 AND discount <= 100)
);


-- Query 1: Show the total sales (total quantity sold and total dollar value) for each customer. (If customer C has made no purchases, still output C, with 0 quantity and dollars).

SELECT 
    C.id,
    C.fname,
    C.lname,
    COALESCE(SUM(S.quantity), 0) AS total_quantity, 						-- Use coalesce so customers with no sales return 0 instead of null
    ROUND(COALESCE(SUM(S.quantity * P.price * (1 - S.discount/100)), 0), 2) AS total_sales	-- Caclulate total dollars spent/customer
FROM customers AS C
LEFT JOIN sales AS S ON C.id = S.cid
LEFT JOIN products AS P ON S.pid = P.id
GROUP BY C.id, C.fname, C.lname
ORDER BY total_sales DESC;


-- Query 2: Show the total sales (total quantity sold and total dollar value) for each state. (If a state has 0 sales, list it explicitly as such in the output).

SELECT 
    ST.sid,
    ST.stname,
    COALESCE(SUM(S.quantity), 0) AS total_quantity,
    ROUND(COALESCE(SUM(S.quantity * P.price * (1 - S.discount/100)), 0), 2) AS total_sales
FROM states AS ST
LEFT JOIN customers AS C ON ST.sid = C.state
LEFT JOIN sales AS S ON C.id = S.cid
LEFT JOIN products AS P ON S.pid = P.id
GROUP BY ST.sid, ST.stname
ORDER BY total_sales DESC;


-- Query 3: Show the total sales for each product, for a given customer. Only products that were actually bought by the given customer are listed. Order by dollar value. It is fine if your query hardcodes
-- a specific customer id (full points). Better would be to write a parameterized query (postgres function) that takes the customer id (cust_id) as argument. Also great: list for EVERY customer.
-- The output schema should be (pid, cust_id, total).

-- Option 1 - Hardcoded customer id
SELECT 
    P.id AS pid,
    4 AS cust_id,									-- Change hardcoded # to desired customer id
    ROUND(SUM(S.quantity * P.price * (1 - S.discount/100)), 2) AS total_sales		-- Don't need coalesce since we don't care about zeroes
FROM products AS P
JOIN sales AS S ON P.id = S.pid
WHERE S.cid = 4										-- Change hardcoded # to desired customer id
GROUP BY P.id
ORDER BY total_sales DESC;

-- Option 2 - List data for all customers
SELECT 
    P.id AS pid,
    S.cid AS cust_id,
    ROUND(SUM(S.quantity * P.price * (1 - S.discount/100)), 2) AS total_sales
FROM products AS P
JOIN sales AS S ON P.id = S.pid
GROUP BY P.id, S.cid
ORDER BY S.cid, total_sales DESC;


-- Query 4: Show the total sales for each product and customer. Order by dollar value. Compared to 3. you will return all tuples 3. returns, plus also show entries for customers C and
-- products P such that C did not buy P (list C, P with 0 total sales).

SELECT 
    P.id AS pid,
    C.id AS cust_id,
    ROUND(COALESCE(SUM(S.quantity * P.price * (1 - S.discount/100)), 0), 2) AS total_sales
FROM products AS P
CROSS JOIN customers AS C			-- Used cross join to create every possible product/customer tuple
LEFT JOIN sales AS S ON P.id = S.pid AND C.id = S.cid
GROUP BY P.id, C.id
ORDER BY total_sales DESC;


-- Query 5: Show the total sales for each product category and state. The output schema should be (category id, state).

SELECT 
    CAT.id AS category_id,
    ST.sid AS state_id,
    ROUND(SUM(S.quantity * P.price * (1 - S.discount/100)), 2) AS total_sales
FROM sales AS S
JOIN products AS P ON S.pid = P.id		-- don't need left or cross join because I don't care about categories/states with 0 sales
JOIN categories AS CAT ON P.cid = CAT.id
JOIN customers AS C ON S.cid = C.id
JOIN states AS ST ON C.state = ST.sid
GROUP BY CAT.id, ST.sid
ORDER BY CAT.id, total_sales DESC;


-- Query 6: For each one of the top 20 product categories (by total revenue) and top 20 customers (by total purchase revenue), return a tuple (top product category ID, top customer ID, quantity sold,
-- dollar value). It is possible that a top-20 customer spent $0 on a top-20 category. List this fact explicitly in the output: (id of “comic books”, id of “jane”, 0, 0) is possible. Extra credit if you can
-- list the rank of the customer and of the category: (cat_id, cat_rank, cust_id, cust_rank, quantity, dollar value).

WITH top_categories AS (		-- Find top 20 categories by total revenue
    SELECT CAT.id AS category_id
    FROM categories AS CAT
    JOIN products AS P ON CAT.id = P.cid
    JOIN sales AS S ON P.id = S.pid
    GROUP BY CAT.id
    ORDER BY SUM(S.quantity * P.price * (1 - S.discount/100)) DESC
    LIMIT 20
),
top_customers AS (			-- Find top 20 customers by total spend
    SELECT C.id AS customer_id
    FROM customers AS C
    JOIN sales AS S ON C.id = S.cid
    JOIN products AS P ON S.pid = P.id
    GROUP BY C.id
    ORDER BY SUM(S.quantity * P.price * (1 - S.discount/100)) DESC
    LIMIT 20
),
category_ranks AS (			-- Rank the top 20 categories. Use dense rank so ranks don't skip
    SELECT 
        TC.category_id,
        DENSE_RANK() OVER (ORDER BY SUM(S.quantity * P.price * (1 - S.discount/100)) DESC) AS cat_rank
    FROM top_categories AS TC
    JOIN products AS P ON TC.category_id = P.cid
    JOIN sales AS S ON P.id = S.pid
    GROUP BY TC.category_id
),
customer_ranks AS (			-- Rank the top 20 customers. Use dense rank so ranks don't skip
    SELECT 
        TCU.customer_id,
        DENSE_RANK() OVER (ORDER BY SUM(S.quantity * P.price * (1 - S.discount/100)) DESC) AS cust_rank
    FROM top_customers AS TCU
    JOIN sales AS S ON TCU.customer_id = S.cid
    JOIN products AS P ON S.pid = P.id
    GROUP BY TCU.customer_id
)
SELECT 					-- Create list of all 400 possible combos  
    CR.category_id,
    CR.cat_rank,
    CUR.customer_id,
    CUR.cust_rank,
    COALESCE(SUM(S.quantity), 0) AS quantity_sold,
    ROUND(COALESCE(SUM(S.quantity * P.price * (1 - S.discount/100)), 0), 2) AS dollar_value
FROM category_ranks AS CR
CROSS JOIN customer_ranks AS CUR
LEFT JOIN products AS P ON CR.category_id = P.cid
LEFT JOIN sales AS S ON P.id = S.pid AND CUR.customer_id = S.cid
GROUP BY CR.category_id, CR.cat_rank, CUR.customer_id, CUR.cust_rank
ORDER BY dollar_value DESC;
