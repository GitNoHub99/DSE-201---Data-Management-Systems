------- MILESTONE 1 ------------------------------------------------------------------------------------------------

-- Create tables
CREATE TABLE users (
    uid INTEGER PRIMARY KEY,
    fname TEXT NOT NULL,
    lname TEXT NOT NULL
);

CREATE TABLE videos (
    vid INTEGER PRIMARY KEY,
    vname TEXT NOT NULL
);

CREATE TABLE logins (
    id INTEGER PRIMARY KEY,
    uid INTEGER NOT NULL REFERENCES users(uid),
    login_time TIMESTAMP NOT NULL
);

CREATE TABLE suggestions (
    id INTEGER PRIMARY KEY,
    login_id INTEGER NOT NULL REFERENCES logins(id),
    vid INTEGER NOT NULL REFERENCES videos(vid)
);

CREATE TABLE likes (
    uid INTEGER NOT NULL REFERENCES users(uid),
    vid INTEGER NOT NULL REFERENCES videos(vid),
    PRIMARY KEY (uid, vid)
);

CREATE TABLE watches (
    id INTEGER PRIMARY KEY,
    uid INTEGER NOT NULL REFERENCES users(uid),
    vid INTEGER NOT NULL REFERENCES videos(vid),
    watch_time TIMESTAMP NOT NULL
);

CREATE TABLE friendships (
    uid1 INTEGER NOT NULL REFERENCES users(uid),
    uid2 INTEGER NOT NULL REFERENCES users(uid),
    PRIMARY KEY (uid1, uid2)
);

-- Query 1: “Overall Likes”: The Top-10 cat videos are the ones that have collected the highest numbers of likes, overall.

SELECT 
    V.vid,
    V.vname,
    COUNT(L.uid) AS like_count
FROM videos AS V
JOIN likes AS L ON V.vid = L.vid
WHERE V.vid NOT IN (					-- excludes videos that user has already liked
    SELECT vid FROM likes WHERE uid = 4744		-- change user id here to see different results
    UNION						-- excludes videos that user has already watched
    SELECT vid FROM watches WHERE uid = 4744		-- enter desired user
)
GROUP BY V.vid, V.vname
ORDER BY like_count DESC
LIMIT 10;


-- Query 2: Option “Friend Likes”: The Top-10 cat videos are the ones that have collected the highest numbers of likes from the friends of X.

SELECT 
    V.vid,
    V.vname,
    COUNT(L.uid) AS friend_like_count
FROM videos AS V
JOIN likes AS L ON V.vid = L.vid
JOIN friendships AS F ON L.uid = F.uid2			-- join on friend's likes of same vids
WHERE F.uid1 = 2862					-- enter desired user
  AND V.vid NOT IN (
      SELECT vid FROM likes WHERE uid = 2862		-- enter desired user, excludes vids they already liked
  )
GROUP BY V.vid, V.vname
ORDER BY friend_like_count DESC
LIMIT 10;


-- Query 3: Option “Friends-of-Friends Likes”: The Top-10 cat videos are the ones that have collected the highest numbers of likes from friends and friends-of-friends.

SELECT 
    V.vid,
    V.vname,
    COUNT(DISTINCT L.uid) AS fof_like_count			-- want unique user likes, use distinct
FROM videos AS V
JOIN likes AS L ON V.vid = L.vid
WHERE L.uid IN (
    SELECT F.uid2 FROM friendships AS F WHERE F.uid1 = 2682	-- enter desired user, direct friends
    UNION															
    SELECT F2.uid2 						-- friends of friends
    FROM friendships AS F1
    JOIN friendships AS F2 ON F1.uid2 = F2.uid1
    WHERE F1.uid1 = 2682
)
AND V.vid NOT IN (
    SELECT vid FROM likes WHERE uid = 2682  			-- enter desired user, exclude vids user already liked
)
GROUP BY V.vid, V.vname
ORDER BY fof_like_count DESC
LIMIT 10;


-- Query 4: Option “My kind of cats”: The Top-10 cat videos are the ones that have collected the most likes from users who have liked at least one cat video that was liked by X.

SELECT 
    V.vid,
    V.vname,
    COUNT(L.uid) AS like_count
FROM videos AS V
JOIN likes AS L ON V.vid = L.vid
WHERE L.uid IN (
    SELECT DISTINCT L2.uid		    	-- find users who liked <= 1 vid that user X also liked
    FROM likes AS L1
    JOIN likes AS L2 ON L1.vid = L2.vid		-- match on same video for other user
    WHERE L1.uid = 1 AND L2.uid != 1		-- enter desired user, exclude user X
)
AND V.vid NOT IN (
    SELECT vid FROM likes WHERE uid = 1  	-- enter desired user, exclude vids user X already liked
)
GROUP BY V.vid, V.vname
ORDER BY like_count DESC
LIMIT 10;


-- Query 5: Option “My kind of cats – with preference (to cat aficionados that have the same tastes)”: The Top-10 cat videos are the ones that have collected the highest sum of weighted likes from every
-- other user Y (i.e., given a cat video, each like on it by user Y is multiplied by a weight assigned to user Y).
-- Weighted recommendations using log cosine similarity lc(X,Y) = log(1 + sum of (X[i] * Y[i]) for all videos i)

WITH user_similarity AS (
    SELECT 					-- calclulate log/cos similarity b/w user 1 & all other users. 
        L2.uid AS other_user,
        LN(1 + COUNT(*)) AS similarity_score	-- similarity = log(1+ count of shared liked vids)
    FROM likes AS L1				-- user 1 likes
    JOIN likes AS L2 ON L1.vid = L2.vid  	-- vids user 1 liked that other people liked
    WHERE L1.uid = 50  				-- enter desired user
      AND L2.uid != 50  			-- enter desired user, filter out from L2
    GROUP BY L2.uid
)
SELECT 
    V.vid,
    V.vname,
	-- for each vid, get similarity score of all users who also liked. Higher scores forr vids liked by similar users
	ROUND(CAST(SUM(US.similarity_score) AS NUMERIC), 4) AS weighted_like_score
FROM videos AS V
JOIN likes AS L ON V.vid = L.vid				-- get users who liked this video
JOIN user_similarity AS US ON L.uid = US.other_user		-- weight by similiarity 
WHERE V.vid NOT IN (						-- exclude videos user1 already liked
    SELECT vid FROM likes WHERE uid = 50  			-- enter desired user, exclude videos user X already liked
)
GROUP BY V.vid, V.vname
ORDER BY weighted_like_score DESC
LIMIT 10;

------- MILESTONE 4 ------------------------------------------------------------------------------------------------

-- pre compute top 100k most liked videos
CREATE TABLE popular_videos AS
SELECT 
    V.vid,
    V.vname,
    COUNT(L.uid) AS like_count
FROM videos AS V
JOIN likes AS L ON V.vid = L.vid
GROUP BY V.vid, V.vname
ORDER BY like_count DESC
LIMIT 100000;


-- Create indexes on precomputed tables
CREATE INDEX idx_popular_vid ON popular_videos(vid);


-- New query 5 using precomputed tables
WITH user_similarity AS (
    SELECT 
        L2.uid AS other_user,
        LN(1 + COUNT(*)) AS similarity_score
    FROM likes AS L1
    JOIN likes AS L2 ON L1.vid = L2.vid
    WHERE L1.uid = 1 AND L2.uid != 1
    GROUP BY L2.uid
)
SELECT 
    PV.vid,
    PV.vname,
    ROUND(CAST(SUM(US.similarity_score) AS NUMERIC), 4) AS weighted_like_score
FROM popular_videos AS PV  -- Use precomputed popular videos instead of full videos table
JOIN likes AS L ON PV.vid = L.vid
JOIN user_similarity AS US ON L.uid = US.other_user
WHERE PV.vid NOT IN (SELECT vid FROM likes WHERE uid = 1)
GROUP BY PV.vid, PV.vname
ORDER BY weighted_like_score DESC
LIMIT 10;

Notes:  Query 5 calculated user similarity by doing very expensive self joins on the likes table (that has over 10M rows) to find shared videos between users, then it aggregated the weighted scores across all videos.
	Once I created the precomputed table that had the top 100k most liked videos, the query scans a smaller dataset instead of the full 1M video table. This gets rid of the expensive sequential scan on videos table which was the main bottleneck.
	This reduced the execution time from 79.9ms to 26.6ms. Maintenance cost for rebuilding the new table when likes data changes is 1.53s. Break even = 28.9. After running new query 5 about 29 times, the precomputation cost is recovered. 
	For a video recommendation platform with medium to high query volume this is achievable. But if data is updated very frequently like hourly, the indexed approach from milestone 3 might be more practical.
