-- Added timestamp for when a video is liked

CREATE TABLE users (
    uid INTEGER PRIMARY KEY,
    fname TEXT NOT NULL,
    lname TEXT NOT NULL
);

CREATE TABLE videos (
    vid INTEGER PRIMARY KEY,
    vname TEXT NOT NULL
);

CREATE TABLE logins (
    id INTEGER PRIMARY KEY,
    uid INTEGER NOT NULL REFERENCES users(uid),
    login_time TIMESTAMP NOT NULL
);

CREATE TABLE suggestions (
    id INTEGER PRIMARY KEY,
    login_id INTEGER NOT NULL REFERENCES logins(id),
    vid INTEGER NOT NULL REFERENCES videos(vid)
);

CREATE TABLE likes (
    uid INTEGER NOT NULL REFERENCES users(uid),
    vid INTEGER NOT NULL REFERENCES videos(vid),
    like_time TIMESTAMP NOT NULL,
    PRIMARY KEY (uid, vid)
);

CREATE TABLE watches (
    id INTEGER PRIMARY KEY,
    uid INTEGER NOT NULL REFERENCES users(uid),
    vid INTEGER NOT NULL REFERENCES videos(vid),
    watch_time TIMESTAMP NOT NULL
);

CREATE TABLE friendships (
    uid1 INTEGER NOT NULL REFERENCES users(uid),
    uid2 INTEGER NOT NULL REFERENCES users(uid),
    PRIMARY KEY (uid1, uid2)
);
